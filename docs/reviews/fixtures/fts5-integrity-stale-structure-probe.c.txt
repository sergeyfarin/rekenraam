/* Standalone reproduction for #403 / #541: PRAGMA integrity_check on a reused
 * connection trusts that connection's cached FTS5 structure. After another
 * connection writes the FTS5 table, the reused connection reports
 *   fts5: checksum mismatch for table "t"
 * for a file that a fresh connection, and FTS5's own 'integrity-check'
 * command, report as ok.
 *
 * Usage: probe [wal|delete] [refresh]
 *   refresh  read one row of t on the reader before each integrity_check
 *            (the application's refreshFTS5StructuresTx workaround).
 * Exit 0: every reader check on the undamaged file says ok, and the damage
 *         control (a deleted t_docsize row) is still reported.
 * Exit 1: a false finding, or the damage control was missed.
 *
 * Results on 2026-11-08 (official amalgamations, both journal modes):
 *   3.53.4  no refresh: exit 1 (run 2 and 3 mismatch)   refresh: exit 0
 *   3.54.1  no refresh: exit 1 (run 2 and 3 mismatch)   refresh: exit 0
 *   3.54.1 + candidate patch (fts5NewTransaction() at the top of
 *          fts5IntegrityMethod, as fts5BeginMethod does): exit 0 without refresh
 *
 * Compile once per official amalgamation, e.g.
 *   cc -O1 -DSQLITE_THREADSAFE=1 -DSQLITE_ENABLE_FTS5 \
 *     -I DIR DIR/sqlite3.c probe.c -o probe -lm -lpthread -ldl
 * It writes probe.db* in the current directory. This is not a production
 * driver, migration or application test replacement.
 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include "sqlite3.h"

static const char *one(sqlite3 *db, const char *sql) {
  static char buf[4096];
  sqlite3_stmt *s;
  buf[0] = 0;
  if (sqlite3_prepare_v2(db, sql, -1, &s, 0) != SQLITE_OK) {
    snprintf(buf, sizeof buf, "prepare error: %s", sqlite3_errmsg(db));
    return buf;
  }
  int rc = sqlite3_step(s);
  if (rc == SQLITE_ROW && sqlite3_column_text(s, 0))
    snprintf(buf, sizeof buf, "%s", (const char *)sqlite3_column_text(s, 0));
  else if (rc != SQLITE_ROW && rc != SQLITE_DONE)
    snprintf(buf, sizeof buf, "step error: %s", sqlite3_errmsg(db));
  while (rc == SQLITE_ROW) rc = sqlite3_step(s);
  sqlite3_finalize(s);
  return buf;
}

static void exec(sqlite3 *db, const char *sql) {
  char *err = 0;
  if (sqlite3_exec(db, sql, 0, 0, &err) != SQLITE_OK) {
    fprintf(stderr, "%s: %s\n", sql, err);
    exit(2);
  }
}

static const char *check(sqlite3 *r, int refresh) {
  if (refresh) one(r, "SELECT rowid FROM t LIMIT 1");
  return one(r, "PRAGMA integrity_check");
}

int main(int argc, char **argv) {
  const char *mode = argc > 1 ? argv[1] : "wal";
  int refresh = argc > 2 && strcmp(argv[2], "refresh") == 0;
  int failed = 0;
  char sql[128];
  unlink("probe.db"); unlink("probe.db-wal"); unlink("probe.db-shm"); unlink("probe.db-journal");
  printf("sqlite %s journal_mode=%s refresh=%d\n", sqlite3_libversion(), mode, refresh);

  sqlite3 *w, *r, *f;
  sqlite3_open("probe.db", &w);
  snprintf(sql, sizeof sql, "PRAGMA journal_mode=%s", mode);
  exec(w, sql);
  exec(w, "CREATE VIRTUAL TABLE t USING fts5(x);"
          "INSERT INTO t(x) VALUES('alpha beta');");
  sqlite3_open("probe.db", &r);

  for (int run = 1; run <= 3; run++) {
    const char *got = check(r, refresh);
    printf("reader integrity_check run %d: %s\n", run, got);
    if (strcmp(got, "ok") != 0) failed = 1;
    snprintf(sql, sizeof sql, "INSERT INTO t(x) VALUES('gamma delta %d')", run);
    exec(w, sql);
  }

  sqlite3_open("probe.db", &f);
  printf("fresh connection integrity_check: %s\n", one(f, "PRAGMA integrity_check"));
  sqlite3_close(f);
  printf("reader FTS5 'integrity-check' command: %s\n",
         sqlite3_exec(r, "INSERT INTO t(t) VALUES('integrity-check')", 0, 0, 0) == SQLITE_OK
             ? "ok" : sqlite3_errmsg(r));

  /* Damage control: real FTS5 damage must still be reported on the reader. */
  exec(w, "DELETE FROM t_docsize WHERE id = 2");
  const char *damaged = check(r, refresh);
  printf("reader integrity_check after deleting a t_docsize row: %s\n", damaged);
  if (strcmp(damaged, "ok") == 0) failed = 1;

  sqlite3_close(r);
  sqlite3_close(w);
  printf("%s\n", failed ? "FAIL" : "PASS");
  return failed;
}
